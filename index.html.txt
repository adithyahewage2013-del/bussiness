function runPayment(v){
  /* ---- DEMO SIMULATION ------------------------------------
     This only runs when the OWNER has explicitly enabled
     "Demo Payment Mode" in the Admin panel. It mimics the
     provider round-trip so the UX can be tested end-to-end.
     It charges nothing and verifies nothing. In production
     this entire function is replaced by a redirect to your
     backend-created Stripe Checkout / PayHere session, with
     access granted ONLY by a verified webhook. */
  $("#payBody").innerHTML = `
    <div class="pay-status">
      <div class="spinner"></div>
      <p style="color:var(--muted);font-size:13px">Contacting payment provider…<br><small>Do not close this window.</small></p>
    </div>`;
  openModal("payModal");
  setTimeout(() => {
    $("#payBody").innerHTML = `
      <div class="pay-status">
        <div class="ico">✅</div>
        <h4 style="margin-bottom:6px">Demo Payment Verified</h4>
        <p style="color:var(--muted);font-size:12.5px;line-height:1.6">
          <b style="color:var(--gold)">SIMULATED</b> — no real charge occurred.<br>
          In production this confirmation arrives from a Stripe/PayHere webhook.
        </p>
        <button class="btn btn-primary btn-block" id="payDoneBtn" style="margin-top:18px">▶ Start Watching</button>
      </div>`;
    purchases.push({ videoId:v.id, title:v.title, price:v.price, date:new Date().toISOString(), status:"simulated" });
    save(LS.purchases, purchases);
    renderAll();
    $("#payDoneBtn").addEventListener("click", () => { closeModal("payModal"); openPlayer(v.id); });
  }, 2200);
}

$("#videoForm").addEventListener("submit", e => {
  e.preventDefault();
  const title = $("#fTitle").value.trim();
  const price = parseFloat($("#fPrice").value);
  const url   = $("#fUrl").value.trim();
  if (!title || isNaN(price)){ toast("Title and a valid price are required.", "err"); return; }
  if (!url){ toast("Provide a video URL or select a file.", "err"); return; }
  const data = {
    title, price: Math.max(0, price), url,
    desc: $("#fDesc").value.trim(), cat: $("#fCat").value,
    dur: $("#fDur").value.trim(), thumb: $("#fThumb").value.trim(),
    emoji: $("#fTitle").value.trim().charAt(0).toUpperCase() || "🎬"
  };
  if (editingId){
    const i = videos.findIndex(v => v.id === editingId);
    if (i > -1) videos[i] = { ...videos[i], ...data };
    toast("Video updated.", "ok");
  } else {
    videos.unshift({ id: uid(), ...data });
    toast("Video added to library.", "ok");
  }
  save(LS.videos, videos); resetForm(); renderAdmin(); renderAll();
});
$("#adminList").addEventListener("click", e => {
  const del = e.target.closest("[data-del]"), ed = e.target.closest("[data-edit]");
  if (del){
    const v = videos.find(x => x.id === del.dataset.del);
    if (confirm(`Delete "${v.title}"? Purchases already made are kept in the purchases list.`)){
      videos = videos.filter(x => x.id !== del.dataset.del);
      save(LS.videos, videos); renderAdmin(); renderAll(); toast("Video deleted.", "ok");
    }
  }
  if (ed){
    const v = videos.find(x => x.id === ed.dataset.edit); if (!v) return;
    editingId = v.id;
    $("#fId").value = v.id; $("#fTitle").value = v.title; $("#fDesc").value = v.desc || "";
    $("#fCat").value = v.cat; $("#fPrice").value = v.price; $("#fDur").value = v.dur || "";
    $("#fThumb").value = v.thumb || ""; $("#fUrl").value = v.url;
    $("#formTitle").textContent = "Edit: " + v.title;
    $("#videoForm").scrollIntoView({behavior:"smooth", block:"center"});
  }
});
